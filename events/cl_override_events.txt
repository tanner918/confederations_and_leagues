#Handles effects of offer confederation and assigns events
mpo_interactions_events.0001 = {
	id_override_priority = 1 # C&L: Setting this definition to override the base game's
	hidden = yes

	trigger = {
		scope:actor = {
			is_ruler = yes
			is_landed = yes
		}
		scope:recipient = {
			is_ruler = yes
			is_landed = yes
		}
	}

	immediate = {
		#Confederation doesn't exist and actor is becoming liege
		if = {
			limit = {
				scope:actor = {
					is_confederation_member = no
				}
			}
			scope:actor = {
				CL_determine_name = yes # C&L: The base game determines whether the confederation being made will be culture-based or named after a duchy here. Because of the Confederation/League split, I moved the naming and creation of the confederation into its own scripted_effect here
				scope:recipient = {
					save_scope_as = other_confederate
				}
				trigger_event = {
					id = mpo_interactions_events.0002
					days = 1
				}
			}
			scope:new_confederation = {
				add_confederation_member = scope:recipient
			}
		}
		#Confederation already exists
		else = {
			scope:actor.confederation = {
				add_confederation_member = scope:recipient
			}
			#letter event
			scope:actor = {
				trigger_event = {
					id = mpo_interactions_events.0003
					days = 1
				}
			}
		}
		#Vassal event
		scope:actor = {
			save_scope_as = other_confederate
		}
		scope:recipient = {
			trigger_event = {
				id = mpo_interactions_events.0002
				days = 1
			}
		}
	}
}

#Confederation event
mpo_interactions_events.0002 = {
	id_override_priority = 1 # C&L: Setting this definition to override the base game's
	type = character_event
	theme = vassal
	title = mpo_interactions_events.0002.t
	desc = {
		desc = mpo_interactions_events.0002.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:new_confederation
				}
				desc = mpo_interactions_events.0002.desc_new
			}
			desc = mpo_interactions_events.0002.desc_joining
		}
	} 
	override_background = {
		reference = bp1_bonfire
	}
	override_effect_2d = {
		reference = smoke
	}
	left_portrait = {
		character = root
		animation = marshal_dagger
		camera = camera_event_very_left
	}
	right_portrait = {
		character = scope:other_confederate
		animation = marshal_dagger
		camera = camera_event_very_right
	}
	immediate = {
		confederation = {
			save_scope_as = confederation
		}
		if = {
			limit = {
				exists = scope:new_confederation
			}
			custom_tooltip = new_confederation_created_tt
		}
		else = {
			custom_tooltip = joined_confederation_tt
		}
		# C&L: Made tooltips shown here properly reflect what type of confederation/league you are now in
		if = {
			limit = {
				confederation = { confederation_type = confederation_type:defensive_league_type }
			}
			custom_tooltip = league_defensive_wars_tt
			custom_tooltip = unlock_leave_league_interaction_tt
			custom_tooltip = league_raiding_attacking_tt
		}
		else = {
			custom_tooltip = confederation_defensive_wars_tt
			custom_tooltip = unlock_leave_confederation_decision_tt
			custom_tooltip = confederation_raiding_attacking_tt
		}
		show_as_tooltip = {
			if = {
				limit = {
					government_has_flag = government_is_nomadic
				}
				add_character_modifier = {
					modifier = mpo_confederation_member_modifier
					years = 5
				}
				capital_county ?= {
					change_county_fertility = major_county_fertility_level_gain
				}
			}
		}
		# C&L: Base game only checks if you are Nomadic, and uses a fallback for tribal. I changed it to instead check based off the respective realm law you use
		CL_federation_realm_law_title_creation_restriction_tooltip_effect = yes
		if = {
			limit = {
				government_has_flag = government_is_nomadic
			}
			custom_tooltip = confederation_migrating_leaving_warning_tt
		}
	}
	#Accept
	option = {
		name = mpo_interactions_events.0002.a
	}
}

	id_override_priority = 1 # C&L: Setting this definition to override the base game's
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:leaving_trade_league ?= yes }
				desc = mpo_decisions_events.0002.t_league
			}
			# C&L: Added specific title for Defensive Leagues
			triggered_desc = {
				trigger = { scope:old_CL_type ?= flag:defensive_league }
				desc = mpo_decisions_events.0002.t_defensive_league
			}
			desc = mpo_decisions_events.0002.t # C&L: Fallsback to assuming you were in a Confederation
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:leaving_trade_league ?= yes }
				desc = mpo_decisions_events.0002.desc_league
			}
			desc = mpo_decisions_events.0002.desc
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:leaving_trade_league ?= yes
					exists = scope:confederation
					scope:confederation = {
						any_confederation_member = {
							count >= 2
							NOT = {
								this = root
							}
						}
					}
				}
				desc = mpo_decisions_events.0002.desc_league_survives
			}
			triggered_desc = {
				trigger = {
					scope:old_CL_type ?= flag:confederation # C&L: Checking if what you are leaving is a confederation
					exists = scope:confederation
					scope:confederation = {
						any_confederation_member = {
							count >= 2
							NOT = {
								this = root
							}
						}
					}
				}
				desc = mpo_decisions_events.0002.desc_confederation_survives
			}
			# C&L: Added this option in case you were in a Defensive League and it remains
			triggered_desc = {
				trigger = {
					scope:old_CL_type ?= flag:defensive_league # C&L: Checking if what you are leaving is a defensive league
					exists = scope:confederation
					scope:confederation = {
						any_confederation_member = {
							count >= 2
							NOT = {
								this = root
							}
						}
					}
				}
				desc = mpo_decisions_events.0002.desc_defensive_league_survives
			}
			triggered_desc = {
				trigger = { scope:leaving_trade_league ?= yes }
				desc = mpo_decisions_events.0002.desc_league_done
			}
			# C&L: Added this option in case you were in a Defensive League and it disbanded
			triggered_desc = {
				trigger = {
					scope:other_confederate = { is_confederation_member = no }
					scope:old_CL_type ?= flag:defensive_league # C&L: Checking if what you are leaving is a defensive league
				}
				desc = mpo_decisions_events.0002.desc_defensive_league_done
			}
			desc = mpo_decisions_events.0002.desc_confederation_done # C&L: Fallsback to assuming you were in a Confederation and it disbanded
		}
	}
	theme = vassal
	left_portrait = {
		character = root
		animation = dismissal
	}
	right_portrait = {
		character = scope:other_confederate
		animation = disappointed
		camera = camera_event_right_forward
	}

	immediate = {
		ordered_in_list = {
			order_by = max_military_strength
			list = confederation_members
			save_scope_as = other_confederate
		}
		scope:confederation = {
			if = {
				limit = { confederation_type = confederation_type:trade_league }
				save_scope_value_as = {
					name = leaving_trade_league
					value = yes
				}
			}
			leave_confederation_membership_effect = { LEAVER = root }
		}
		if = {
			limit = {
				government_has_flag = government_is_nomadic
			}
			show_as_tooltip = {
				remove_character_modifier = mpo_confederation_member_modifier
			}
		}
		if = {
			limit = { scope:leaving_trade_league ?= yes }
			custom_tooltip = trade_league_member_lose_opinion_tt
			hidden_effect = {
				every_in_list = {
					list = confederation_members
					limit = { is_alive = yes }
					add_opinion = {
						modifier = turncoat_opinion
						target = root
						opinion = -20
					}
				}
			}
		}
		else = {
			every_in_list = {
				list = confederation_members
				limit = { is_alive = yes }
				custom = custom.every_confederation_member
				add_opinion = {
					modifier = turncoat_opinion
					target = root
					opinion = -20
				}
			}
		}
	}

	#Bye
	option = {
		name = mpo_decisions_events.0002.a
		every_in_list = {
			list = confederation_members
			trigger_event = mpo_decisions_events.0003
		}
	}
}

scripted_trigger mpo_greatest_of_khans_0002_confederation_trigger = {
	is_confederation_member = yes
	confederation = { # C&L: Adding this condition that makes sure that the confederation/league the neighboring ruler is in is actually joinable
		OR = {
			AND = {
				confederation_type = confederation_type:nomadic_confederation
				scope:searching_for_confed = { CL_uses_confederations = yes }
			}
			AND = {
				confederation_type = confederation_type:defensive_league_type
				scope:searching_for_confed = { CL_uses_leagues = yes }
			}
		}
	}
	#Prev be with you
